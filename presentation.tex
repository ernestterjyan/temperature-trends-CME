\section{Conclusions}
\begin{frame}{Later warming, qualified evidence of a break}
\vspace{2mm}
\begin{block}{A positive later fitted trend}
\small $+\PostSlope$\,\degC\ per decade; both interval constructions exclude zero.\par
The early slope's intervals include zero.
\end{block}
\begin{block}{Break evidence depends on the bandwidth}
\small Best-fitting date: \BreakYear. The primary DWB test rejects at 5\%;\par
bandwidths 12--15 do not. The date remains an uncertain model feature.
\end{block}
\begin{block}{Dependence matters for inference}
\small Independent methods over-reject in the AR scenarios.\par
DWB avoids that inflation, but remains an approximation.
\end{block}
\source{The later slope averages 1851--2013, not recent decades. Missing years, one station and model assumptions limit broader or causal claims.}
\end{frame}

\begin{frame}{DW and BP use different references}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\small\textbf{DW Monte Carlo}\par\vspace{2mm}
Specify iid Gaussian errors and a fixed design $X$.\par
\[e^{(b)}=M_Xz^{(b)},\quad z^{(b)}\sim N(0,I_n).\]
Coefficients disappear under projection; the error scale cancels in DW.\par\vspace{2mm}
Simulate separate lower and upper tails.
\column{.48\textwidth}
\small\textbf{BP asymptotic reference}\par\vspace{2mm}
Regress $e_i^2$ on $1,x_i$; compute $BP=nR^2$.\par
\[BP\ \overset{a}{\sim}\ \chi^2_1\quad\text{under regularity conditions}.\]
Homoskedasticity alone does not specify the finite-sample distribution: error shape and dependence matter.
\end{columns}
\vspace{4mm}
\point{Gaussian simulation of BP is possible under a fully specified null.}
\source{It is not automatically valid for the broader homoskedasticity null. Squared residuals do not make simulation impossible; additional distributional assumptions are the issue.}

\end{frame}

\begin{frame}{Generate each testing sample under the null}
\small
\[\widetilde e_i=(e_{0i}-\bar e_0)\sqrt{n/(n-2)},\qquad
y_i^*=\widehat y_{0i}+e_i^*.\]
\begin{itemize}
\item \textbf{IID:} draw $e_i^*$ independently with replacement from $\{\widetilde e_j\}$.
\item \textbf{Independent wild:} $e_i^*=\widetilde e_iw_i$, independent equiprobable signs $w_i=\pm1$.
\item \textbf{DWB:} $e_i^*=\widetilde e_iw_i$, with $w_1\sim N(0,1)$ and
\end{itemize}
\vspace{-2mm}
\[a_i=e^{-(s_i-s_{i-1})/\ell},\quad
w_i=a_iw_{i-1}+\sqrt{1-a_i^2}z_i,\quad z_i\overset{\mathrm{iid}}\sim N(0,1).\]
\vspace{-2mm}
\[F_T^*=RSS_0^*-\min_k RSS_k^*,\qquad
p=\frac{1+\sum_b1\{F_{T,b}^*\geq F_T\}}{B+1}.\]
\source{Refit the null and all candidate alternatives in every draw; $B=9,999$. Type-1 95\% critical value, reject for $p\leq0.05$. DWB preserves residual scale and calendar-distance multiplier dependence.}

\end{frame}

\begin{frame}{Testing and intervals use different models}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\textbf{Break test}\par\vspace{2mm}
Null mean $\widehat y_0$ and residuals $e_0$.\par\vspace{2mm}
Scale residuals by $\sqrt{n/(n-2)}$.\par\vspace{2mm}
Repeat the full search and count $F_T^*\geq F_T$.
\column{.48\textwidth}
\textbf{Slope intervals}\par\vspace{2mm}
Alternative mean $\widehat y_1$ and residuals $e_1$.\par\vspace{2mm}
Scale residuals by $\sqrt{n/(n-3)}$.\par\vspace{2mm}
Reselect the date and recompute slopes and standard errors.
\end{columns}
\vspace{5mm}
\[
 J_i=[1,x_i,(x_i-\tau)_+,-\widehat\delta\,1(x_i>\tau)],\qquad
 K_{ij}=\max\{1-|s_i-s_j|/7,0\}
\]
\source{Studentisation includes the local date direction and uses correction $n/(n-4)$. A rescaled indicator replaces the fourth column for numerical stability when the kink is nonzero. Weak identification can limit coverage.}

\end{frame}
